cmake_minimum_required(VERSION 3.16)
project(student_grade_management LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(student_grade_management src/main.cpp)
if(MSVC)
  target_compile_options(student_grade_management PRIVATE /W4)
else()
  target_compile_options(student_grade_management PRIVATE -Wall -Wextra -Wpedantic)
endif()
